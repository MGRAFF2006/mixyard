cmake_minimum_required(VERSION 3.22)
project(mixyard_mobile_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MIXYARD_ENGINE_ROOT "" CACHE PATH "Checkout containing adapters/engine/include/mixyard_host.h")
set(MIXYARD_ENGINE_LIBRARY "" CACHE FILEPATH "Built canonical Rust host library")
if(NOT EXISTS "${MIXYARD_ENGINE_ROOT}/adapters/engine/include/mixyard_host.h" OR NOT EXISTS "${MIXYARD_ENGINE_LIBRARY}")
  message(FATAL_ERROR "Provide the reviewed canonical host checkout and its actual built library")
endif()
find_package(JNI REQUIRED)
add_library(mixyard_mobile_host SHARED jni/host_jni.cpp)
target_include_directories(mixyard_mobile_host PRIVATE ${JNI_INCLUDE_DIRS} "${MIXYARD_ENGINE_ROOT}/adapters/engine/include")
target_link_libraries(mixyard_mobile_host PRIVATE "${MIXYARD_ENGINE_LIBRARY}")
target_compile_options(mixyard_mobile_host PRIVATE -Wall -Wextra -Werror)
