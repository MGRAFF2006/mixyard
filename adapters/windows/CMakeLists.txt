cmake_minimum_required(VERSION 3.20)
project(mixyard_windows_probe LANGUAGES CXX)

include(CTest)
find_package(Threads REQUIRED)
add_executable(mixyard-windows-model-check model_check.cpp)
target_compile_features(mixyard-windows-model-check PRIVATE cxx_std_17)
target_link_libraries(mixyard-windows-model-check PRIVATE Threads::Threads)
add_test(NAME windows-buffer-wav-model COMMAND mixyard-windows-model-check)

if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "The native probe requires MSVC and a Windows SDK >= 10.0.20348.0 (WRL/process loopback).")
  endif()
  if(CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION AND CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION VERSION_LESS "10.0.20348.0")
    message(FATAL_ERROR "Select Windows SDK >= 10.0.20348.0 for process-loopback headers.")
  endif()
  add_executable(mixyard-windows-probe probe.cpp)
  target_compile_features(mixyard-windows-probe PRIVATE cxx_std_17)
  target_compile_definitions(mixyard-windows-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(mixyard-windows-probe PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_libraries(mixyard-windows-probe PRIVATE ole32 uuid mmdevapi avrt)
else()
  message(STATUS "Non-Windows: portable model check only; native probe is NOT built.")
endif()
