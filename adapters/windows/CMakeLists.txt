cmake_minimum_required(VERSION 3.20)
project(mixyard_windows_probe LANGUAGES CXX)

include(CTest)
find_package(Threads REQUIRED)
add_library(mixyard-windows-worker STATIC worker_queue.cpp wasapi_worker.cpp)
target_compile_features(mixyard-windows-worker PUBLIC cxx_std_17)
target_include_directories(mixyard-windows-worker PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(mixyard-windows-worker-check worker_check.cpp)
target_link_libraries(mixyard-windows-worker-check PRIVATE mixyard-windows-worker Threads::Threads)
add_test(NAME windows-exact-worker-model COMMAND mixyard-windows-worker-check)
add_executable(mixyard-windows-model-check model_check.cpp)
target_compile_features(mixyard-windows-model-check PRIVATE cxx_std_17)
target_link_libraries(mixyard-windows-model-check PRIVATE Threads::Threads)
add_test(NAME windows-buffer-wav-model COMMAND mixyard-windows-model-check)

add_executable(mixyard-inventory-check inventory/check.cpp)
target_compile_features(mixyard-inventory-check PRIVATE cxx_std_17)
add_test(NAME native-inventory-protocol COMMAND mixyard-inventory-check)

if(WIN32)
  target_compile_definitions(mixyard-windows-worker PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(mixyard-windows-worker PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  target_compile_definitions(mixyard-windows-worker-check PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(mixyard-windows-worker-check PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  target_link_libraries(mixyard-windows-worker PUBLIC ole32 uuid mmdevapi avrt)
  add_executable(mixyard-windows-worker-sdk-check worker_sdk_check.cpp)
  target_compile_definitions(mixyard-windows-worker-sdk-check PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(mixyard-windows-worker-sdk-check PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  target_link_libraries(mixyard-windows-worker-sdk-check PRIVATE mixyard-windows-worker)
  add_test(NAME windows-worker-sdk-injected COMMAND mixyard-windows-worker-sdk-check)
  target_compile_definitions(mixyard-windows-model-check PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(NOT MSVC)
    message(FATAL_ERROR "The native probe requires MSVC and a Windows SDK >= 10.0.20348.0 (WRL/process loopback).")
  endif()
  if(CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION AND CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION VERSION_LESS "10.0.20348.0")
    message(FATAL_ERROR "Select Windows SDK >= 10.0.20348.0 for process-loopback headers.")
  endif()
  add_executable(mixyard-windows-probe probe.cpp worker_bridge.cpp)
  target_compile_features(mixyard-windows-probe PRIVATE cxx_std_17)
  target_compile_definitions(mixyard-windows-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(mixyard-windows-probe PRIVATE /W4 /permissive- /EHsc /utf-8)
  target_link_libraries(mixyard-windows-probe PRIVATE mixyard-windows-worker Threads::Threads ole32 uuid mmdevapi avrt)

  add_executable(mixyard-windows-inventory inventory/main.cpp)
  target_compile_features(mixyard-windows-inventory PRIVATE cxx_std_17)
  target_compile_definitions(mixyard-windows-inventory PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(mixyard-windows-inventory PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  target_link_libraries(mixyard-windows-inventory PRIVATE ole32 uuid mmdevapi)
  install(TARGETS mixyard-windows-inventory RUNTIME DESTINATION bin)
else()
  message(STATUS "Non-Windows: portable model check only; native probe is NOT built.")
endif()
