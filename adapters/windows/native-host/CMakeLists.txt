cmake_minimum_required(VERSION 3.20)
project(mixyard_windows_native_host LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)

set(MIXYARD_HOST_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../engine/include" CACHE PATH "Shared Rust host ABI include directory")
set(MIXYARD_HOST_LIBRARY "" CACHE FILEPATH "Already-built shared Rust host static library")
if(NOT EXISTS "${MIXYARD_HOST_INCLUDE}/mixyard_host.h" OR NOT EXISTS "${MIXYARD_HOST_LIBRARY}")
  message(FATAL_ERROR "Build adapters/engine first, then provide MIXYARD_HOST_LIBRARY and MIXYARD_HOST_INCLUDE; no fallback mixer exists.")
endif()
add_library(mixyard-shared-host STATIC IMPORTED)
set_target_properties(mixyard-shared-host PROPERTIES IMPORTED_LOCATION "${MIXYARD_HOST_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${MIXYARD_HOST_INCLUDE}")
add_library(mixyard-windows-native-host STATIC bridge.cpp ../worker_queue.cpp ../wasapi_worker.cpp)
target_compile_features(mixyard-windows-native-host PUBLIC cxx_std_17)
target_include_directories(mixyard-windows-native-host PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(mixyard-windows-native-host PUBLIC mixyard-shared-host Threads::Threads)
if(MSVC)
  target_compile_definitions(mixyard-windows-native-host PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_compile_options(mixyard-windows-native-host PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  target_link_libraries(mixyard-windows-native-host PUBLIC ntdll userenv ws2_32 bcrypt advapi32 ole32 uuid mmdevapi avrt)
else()
  target_compile_options(mixyard-windows-native-host PRIVATE -Wall -Wextra -Werror -pedantic)
  target_link_libraries(mixyard-windows-native-host PUBLIC ${CMAKE_DL_LIBS} m)
endif()
add_executable(mixyard-windows-host-check check.cpp)
target_link_libraries(mixyard-windows-host-check PRIVATE mixyard-windows-native-host)
if(MSVC)
  target_compile_options(mixyard-windows-host-check PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  target_link_libraries(mixyard-windows-host-check PRIVATE ole32 uuid)
else()
  target_compile_options(mixyard-windows-host-check PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME windows-linked-rust-host COMMAND mixyard-windows-host-check)
